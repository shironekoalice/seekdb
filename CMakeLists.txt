cmake_minimum_required(VERSION 3.20)
project(minivectordb LANGUAGES CXX)
option(SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_executable(minivectordb src/main.cpp src/sql.cpp src/storage.cpp)
target_compile_features(minivectordb PRIVATE cxx_std_20)
target_include_directories(minivectordb PRIVATE third_party src)
target_compile_options(minivectordb PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(SANITIZE)
  target_compile_options(minivectordb PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(minivectordb PRIVATE -fsanitize=address,undefined)
endif()
install(TARGETS minivectordb RUNTIME DESTINATION bin)
add_executable(mvdb_oracle tools/oracle.cpp)
target_compile_features(mvdb_oracle PRIVATE cxx_std_20)
target_include_directories(mvdb_oracle PRIVATE third_party)
